\begin{frame}{What We Are Building}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{Goal:} A browser extension that intercepts social media posts
      \emph{before sharing} and issues a misinformation nudge if the content
      is likely false or misleading.

      \bigskip
      \textbf{Research arm:} Empirical validation of the Guriev et al.\
      theoretical model of persuasion and belief updating.
      Anonymised behavioural interaction data is collected
      to test the model's predictions.

      \bigskip
      \begin{itemize}
        \item Platforms: Twitter/X, Facebook
        \item Languages: English + French
        \item Deadline: July 2026
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{block}{Key design constraint}
        User-facing latency must be \textbf{under one second.}

        \medskip
        Long-running verification is dispatched
        \emph{asynchronously} — a nudge can fire before
        the full fact-check completes, with results cached
        for future users sharing the same claim.
      \end{block}
    \end{column}
  \end{columns}

% ── 4. Claim Extraction ──────────────────────────────────────
\begin{frame}{Step 1 — Claim Extraction \hfill \normalsize\textcolor{slate}{Qwen3-32B via Groq · 2 sequential LLM calls}}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Call 1 — Detection \& Decomposition}
      \begin{itemize}
        \item Is the post \emph{factually checkable}?\\
              {\footnotesize Decided by \textbf{form}, not plausibility.
              ``The earth is flat'' is checkable.
              Opinions, predictions, and questions are not.}
        \item Decompose into \alert{atomic claims} —\\
              single independently verifiable assertions
        \item Generate \textbf{3 search queries} per claim:\\
              {\footnotesize direct factual · contextual · debunking-oriented}
        \item \textbf{Early exit} if no checkable claims
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Call 2 — Verifying Questions}
      \begin{itemize}
        \item 5--8 \alert{analytical questions} per claim\\
              {\footnotesize Not search queries — the interpretive
              scaffold the verifier reasons over in Step~3.}
        \item Question types:\\
              {\footnotesize
              \quad• Official statements \& attributions\\
              \quad• Expert / scientific consensus\\
              \quad• Contradicting sources\\
              \quad• Quantitative specifics (numbers, dates)\\
              \quad• Temporal anchoring
              }
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ── 5. Evidence Retrieval ────────────────────────────────────
\begin{frame}{Step 2 — Evidence Retrieval}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Three parallel sources}
      \begin{itemize}
        \item \alert{Google Fact Check Tools API}\\
              {\footnotesize Structured verdicts from AFP, PolitiFact, Snopes, \ldots}\\
              {\small\textcolor{forestg}{cred\,=\,0.90}}
        \item \alert{Wikipedia} — top-2 article summaries\\
              {\small\textcolor{steel}{cred\,=\,0.75}}
        \item \alert{Serper web search}\\
              {\footnotesize 3 queries $\times$ 3 URLs, date-capped to claim date}\\
              {\small\textcolor{slate}{cred from CRED-1 (2\,673 domains)}}
      \end{itemize}
      \medskip
      \textbf{Before scraping each URL}
      \begin{itemize}
        \item CRED-1 filter: drop fake / unreliable / cred $<$ 0.30
        \item Structural blocklist: login walls, hard paywalls
        \item \textbf{BM25} — top-2 $\times$ 300-word chunks per page
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Post-retrieval pipeline}
      \medskip
      \begin{enumerate}
        \item \textbf{Cross-encoder rerank}\\
              {\footnotesize \texttt{ms-marco-MiniLM-L6-v2}}\\
              {\footnotesize All items scored vs.\ claim $\to$ keep top 10}
        \smallskip
        \item \textbf{MMR diversification}\\
              {\footnotesize Maximal Marginal Relevance}\\
              {\footnotesize $\lambda = 0.75$ relevance $\;/\;$ $0.25$ diversity}\\
              {\footnotesize $\to$ up to 5 final items to the verifier}
      \end{enumerate}
      \bigskip
      {\footnotesize\textcolor{slate}{Prevents the verifier from receiving\\
      multiple rephrased versions of the same source.}}
    \end{column}
  \end{columns}
\end{frame}

% ── 6. FIRE Verification ─────────────────────────────────────
\begin{frame}{Step 3 — FIRE Iterative Verification \hfill \normalsize\textcolor{slate}{Llama-3.3-70B via Groq\,{\tiny(→ DeepSeek V3)}}}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Each round — \texttt{verify\_claim}}
      \begin{itemize}
        \item Answers each verifying question from evidence,\\
              weighting higher-credibility sources ($\geq 0.6$) more
        \item Rates \alert{all four labels independently} on\\
              a 1--5 Likert scale (rubric on next slide):\\
              {\footnotesize Supported · Refuted · NEI · Conflicting Evidence}
        \item Verdict $=$ argmax Likert\\
              {\footnotesize Ties: Refuted $>$ Supported $>$ CE $>$ NEI}
        \item Returns reasoning trace, justification, key URLs
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{FIRE loop (up to 5 rounds)}
      \begin{enumerate}
        \item Compute \alert{label margin}\\
              {\footnotesize $\text{margin} = \text{top Likert} - \text{2nd Likert}$\\
              High $\to$ model decided; low $\to$ torn between labels}
        \item \textbf{Exit} if verdict $\neq$ NEI and margin $\geq 4$
        \item \textbf{Always continue} if verdict $=$ NEI\\
              {\footnotesize Model declared insufficient evidence}
        \item \textbf{Stop} if follow-up query is redundant\\
              {\footnotesize cosine sim $> 0.9$ vs.\ all past queries}
        \item Otherwise: generate targeted follow-up query\\
              $\to$ retrieve $\to$ dedup-merge $\to$ repeat
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

% ── 7. Likert Rubric ─────────────────────────────────────────
\begin{frame}{Verifier Scoring Rubric — Likert 1--5 Per Label}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{forestg}{Supported}} — ``evidence confirms the claim''\\[3pt]
      {\footnotesize
      \textbf{5:} Two+ credible sources ($\geq 0.6$) confirm including all key qualifiers; no credible contradiction.\\
      \textbf{4:} Multiple sources agree, or one high-cred ($\geq 0.7$) source confirms directly.\\
      \textbf{3:} One credible ($\geq 0.5$) source supports; no corroboration.\\
      \textbf{1--2:} Weak / off-topic support only.
      }

      \bigskip
      \textbf{\textcolor{crimson}{Refuted}} — ``evidence contradicts the claim''\\[3pt]
      {\footnotesize
      \textbf{5:} Two+ credible sources ($\geq 0.6$) contradict a load-bearing qualifier; no credible support.\\
      \textbf{4:} One high-cred ($\geq 0.7$) source directly refutes a key element.\\
      \textbf{3:} One credible ($\geq 0.5$) source contradicts; no corroboration.\\
      \textbf{1--2:} Weak / off-topic contradiction only.
      }
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{slate}{Not Enough Evidence (NEI)}} — ``evidence does not address the claim''\\[3pt]
      {\footnotesize
      \textbf{Must score $\leq 2$} whenever any credible source ($\geq 0.5$) speaks to the claim — even partially.\\
      \textbf{5:} No retrieved item touches the claim at all.\\
      \textbf{3--4:} Topically related sources only; no direct address.\\
      NEI is about \emph{coverage}, not certainty.
      }

      \bigskip
      \textbf{\textcolor{amber}{Conflicting Evidence (CE)}} — ``credible sources disagree''\\[3pt]
      {\footnotesize
      \textbf{Must score $\geq 4$} whenever one credible source supports AND one credible source refutes the same load-bearing element.\\
      \textbf{5:} Two+ high-cred ($\geq 0.6$) sources on each side.\\
      \textbf{3:} One credible / one weak-credibility source disagree.
      }

      \bigskip
      {\footnotesize\textcolor{slate}{\textbf{Tie-break:} If Supported and Refuted both $\geq 3$, raise CE to at least max(Supp, Ref).}}
    \end{column}
  \end{columns}
\end{frame}

% ── 9. Misleadingness Detection ──────────────────────────────
\begin{frame}{Misleadingness Detection — Our Differentiator}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{The problem SOTA systems miss}

      \medskip
      ClaimCheck, FIRE, HerO 2, CTU AIC all target
      \emph{stated factual claims} — they assess whether
      what the post says is true.

      \medskip
      They do not detect posts that are \alert{technically true but misleading.}

      \bigskip
      \textbf{Two categories}\\[4pt]
      \textbf{Misleading by scope:} A broad claim backed by only a narrow instance.\\
      {\footnotesize ``A study shows vaccines cause autism.'' — one retracted paper, not evidence.}

      \medskip
      \textbf{Misleading by implication / omission:} A true statement that naturally
      suggests a false conclusion, or is only true under unstated conditions.\\
      {\footnotesize ``Cities with more immigrants have higher crime rates'' — implies causation; omits
      confounders, time period, and absolute magnitudes.}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{4-step stripping algorithm (per claim A)}

      \medskip
      \begin{enumerate}
        \item Establish A is likely true, preserving all qualifiers
              (time period, sample size, population).
        \item \textbf{Strip qualifiers} $\to \tilde{A}$.
              If $\tilde{A}$ is likely false $\to$ \alert{misleading by scope.}
        \item \textbf{Derive implication} B: the first factual inference
              a reasonable person draws from A.
              If B is likely false $\to$ \alert{misleading by implication.}
        \item \textbf{Augment with silent assumptions} $\to A^+$.
              If $A^+$ is likely false $\to$ \alert{misleading by omission.}
      \end{enumerate}

      \medskip
      Each derived claim ($\tilde{A}$, B, $A^+$) is fed through
      the standard retrieval + verification sub-pipeline.
      A misleading verdict raises the post's falsity score
      even when the literal claim is \emph{Supported.}

      \medskip
      \src{$\to$ Applied after explicit claim verification in Tier~3. Status: designed, not yet implemented.}
    \end{column}
  \end{columns}
\end{frame}

% ── 12. CE Recall ────────────────────────────────────────────
\begin{frame}{Conflicting Evidence Recall = 0\% — Problem \& Fix}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{What happened}

      \medskip
      12 gold-CE claims in the 200-claim run.\\
      Predicted: 6 Refuted, 5 NEI, 1 Supported, \alert{0 CE.}

      Only 6 predictions of CE across the whole sample — all wrong class.

      \bigskip
      \textbf{Why the verifier avoids CE}

      \begin{enumerate}
        \item CE requires the model to find credible evidence
              \emph{on both sides} of the same claim — a retrieval task
              that a single-query strategy rarely satisfies.
        \item Under self-consistency ($\times 3$ runs), averaging Likert scores
              across rounds systematically suppresses CE: if runs 1 and 3
              lean Refuted and run 2 leans CE, the averaged CE score
              rarely becomes the argmax.
      \end{enumerate}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Fix candidates}

      \medskip
      \textbf{1. Majority-vote aggregation}\\
      {\footnotesize Instead of argmax of averaged Likert,
      take the verdict with the most votes across the $n$ SC rounds.
      If two runs say CE and one says Refuted, the verdict is CE.
      Cheap to implement — does not change \texttt{verify\_claim}.}

      \medskip
      \textbf{2. Targeted CE retrieval}\\
      {\footnotesize After an initial Supported or Refuted verdict,
      explicitly generate a follow-up query designed to surface
      \emph{contradicting} evidence. Forces the FIRE loop
      to look for the other side.}

      \medskip
      \textbf{3. Separate CE classifier}\\
      {\footnotesize Fine-tune a small classifier on CE vs.\ non-CE
      cases using retrieved evidence pairs, bypassing the LLM's
      CE-avoidance bias entirely.}

      \medskip
      {\footnotesize\textcolor{slate}{Priority: majority-vote is a one-line change;
      test it first.}}
    \end{column}
  \end{columns}
\end{frame}

% ── 15. KB Routing ───────────────────────────────────────────
\begin{frame}{Cost Lever: KB Routing Before Web Search}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{The problem with the current architecture}\\[4pt]
      {\footnotesize All three retrieval sources — FCTAPI, Wikipedia,
      and Serper — run \emph{in parallel} on every claim, regardless
      of whether a professional fact-check already exists.
      At scale, this means paying for Serper queries on claims that
      could be answered for free.}

      \bigskip
      \textbf{The fix: sequential KB-first routing}
      \begin{enumerate}
        \item Query FCTAPI + vector DB first
              {\footnotesize (~300ms, free)}
        \item If a high-confidence match is found:\\
              return cached verdict, skip Serper entirely,
              FIRE exits after 1 round
        \item If no match:\\
              proceed to full Serper retrieval + FIRE
      \end{enumerate}

      \bigskip
      {\footnotesize\textcolor{slate}{Code change: the parallel
      \texttt{ThreadPoolExecutor} in \texttt{retrieval.py} becomes
      two-phase. Latency cost is negligible in an async Tier~3
      pipeline (already 5--30\,s total).}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{What qualifies as a KB hit?}
      \begin{itemize}
        \item \textbf{FCTAPI match} — a professional fact-checker
              (AFP, PolitiFact, Snopes, \ldots) has already reviewed
              this claim or a semantically close one
        \item \textbf{Vector DB match} — our own growing database of
              previously verified claims (cosine similarity above
              threshold)
      \end{itemize}

      \bigskip
      \textbf{Expected hit rate (mature system)}\\[4pt]
      {\footnotesize For viral political and health misinformation —
      the content our users are most likely to share — professional
      fact-checkers have high coverage. Conservative estimate: \alert{40\%}
      of novel claims (post cache-miss) resolved without Serper.
      Hit rate grows as the vector DB matures.}

      \bigskip
      \textbf{Effect on FIRE}\\[4pt]
      {\footnotesize KB hits carry credibility 0.90. The verifier
      receives a strong, directly relevant source and almost always
      exits after round~1, cutting verification tokens by $\sim$60\%
      for those claims.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Scenario B — Wide Study: Stack \& Compute}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Why caching changes the economics}\\[4pt]
      {\footnotesize Viral misinformation is shared repeatedly — the same
      false claim reaches thousands of users in hours. After the first
      user triggers a full Tier~3 check, every subsequent user gets
      the cached verdict instantly at zero compute cost. At 80\% cache
      hit rate, the effective load drops \alert{5$\times$} from 2M to
      400K novel checks per day.}

      \bigskip
      \textbf{Compute options for self-hosted inference}
      \begin{itemize}
        \item \textbf{HPC allocation} — DRAC (Canada), Jean Zay (France),
              university cluster. GPU-hours at zero marginal cost
              for academic research. \alert{Recommended path.}
        \item \textbf{Cloud GPU cluster} — \$30--50K/month for a
              dedicated vLLM setup ($\sim$50--80$\times$ A100 80GB)
        \item \textbf{Smaller model} — 7B parameter model cuts
              GPU needs $\approx$10$\times$ at some accuracy cost
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Architecture additions vs.\ Scenario A}
      \begin{itemize}
        \item \textbf{Vector DB (Qdrant)} — stores claim embeddings
              and verdicts; cosine similarity lookup on every incoming
              claim before hitting Tier~3
        \item \textbf{Async job queue (Redis + Celery)} — first user
              triggers a background Tier~3 job; result written to
              cache on completion
        \item \textbf{Multi-region load balancer} — 10-country
              deployment requires low-latency routing
        \item \textbf{Observability stack} — latency per tier,
              nudge rate, cache hit rate, per-country dashboards
      \end{itemize}

      \bigskip
      \textbf{Timeline to production:} 4--6 months
    \end{column}
  \end{columns}
\end{frame}
