% ── 2. Goal ──────────────────────────────────────────────────
\begin{frame}{What We Are Building}
  A \textbf{browser extension} that intercepts social media posts
  \emph{before sharing} and issues a misinformation nudge if the
  content is likely false or misleading.

  \bigskip
  \begin{itemize}\setlength{\itemsep}{10pt}
    \item \textbf{Platforms:} Twitter/X and Facebook
    \item \textbf{Languages:} English + French
    \item \textbf{Research arm:} Empirical validation of the Guriev et al.\ persuasion model — anonymised behavioural data collected at scale
    \item \textbf{Key constraint:} User-facing latency under one second; verification is dispatched asynchronously
  \end{itemize}
\end{frame}

% ── 4. Claim Extraction A ────────────────────────────────────
\begin{frame}{Step 1 — Detection \& Decomposition \hfill \normalsize\textcolor{slate}{Qwen3-32B · Call 1}}

  \begin{itemize}\setlength{\itemsep}{14pt}
    \item \textbf{Is this post factually checkable?}\\
          Decided by \emph{form}, not plausibility.
          ``The earth is flat'' is checkable. Opinions, predictions, and questions are not.
    \item \textbf{Decompose into atomic claims}\\
          Each claim is a single, independently verifiable assertion.
    \item \textbf{Generate 3 search queries per claim}\\
          Direct factual · contextual · debunking-oriented
    \item \textbf{Early exit} if no checkable claims found
  \end{itemize}
\end{frame}

% ── 5. Claim Extraction B ────────────────────────────────────
\begin{frame}{Step 1 — Verifying Questions \hfill \normalsize\textcolor{slate}{Qwen3-32B · Call 2}}

  5--8 \alert{analytical questions} per claim — the interpretive scaffold the verifier
  reasons over in Step 3. Not search queries.

  \bigskip
  \begin{itemize}\setlength{\itemsep}{12pt}
    \item Official statements and attributions
    \item Expert or scientific consensus
    \item Contradicting sources
    \item Quantitative specifics — numbers, dates, scope
    \item Temporal anchoring
  \end{itemize}
\end{frame}

% ── 6. Evidence Retrieval A ──────────────────────────────────
\begin{frame}{Step 2 — Evidence Sources \hfill \normalsize\textcolor{slate}{Three parallel streams}}

  \begin{itemize}\setlength{\itemsep}{16pt}
    \item \textbf{\textcolor{forestg}{Google Fact Check Tools API}} \hfill cred\,=\,0.90\\
          Structured verdicts from AFP, PolitiFact, Snopes, and hundreds of regional outlets
    \item \textbf{Wikipedia} \hfill cred\,=\,0.75\\
          Top-2 article summaries for named entities and events in the claim
    \item \textbf{Serper web search} \hfill cred from CRED-1\\
          3 queries $\times$ 3 URLs, date-capped to the claim date\\
          Filtered: fake/unreliable domains (CRED-1 $<$ 0.30) and structural blocklist (paywalls, login walls)
  \end{itemize}
\end{frame}

% ── 7. Evidence Retrieval B ──────────────────────────────────
\begin{frame}{Step 2 — Filtering \& Ranking}

  After scraping, all evidence from the three streams is merged and processed:

  \bigskip
  \begin{itemize}\setlength{\itemsep}{16pt}
    \item \textbf{BM25 chunk extraction}\\
          Top-2 $\times$ 300-word chunks per scraped page, scored against the claim text
    \item \textbf{Cross-encoder reranking} \hfill \texttt{ms-marco-MiniLM-L6-v2}\\
          All items scored versus the claim; top 10 retained
    \item \textbf{MMR diversification} \hfill $\lambda = 0.75$ relevance\,/\,$0.25$ diversity\\
          Selects up to 5 final items; prevents the verifier from receiving
          multiple paraphrases of the same source
  \end{itemize}
\end{frame}

% ── 8. FIRE A ────────────────────────────────────────────────
\begin{frame}{Step 3 — Single Verification Round \hfill \normalsize\textcolor{slate}{Llama-3.3-70B via Groq}}

  \begin{itemize}\setlength{\itemsep}{14pt}
    \item Answers each verifying question from evidence, citing source URLs
    \item Weights higher-credibility sources ($\geq 0.6$) more heavily
    \item Rates \alert{all four labels independently} on a 1--5 Likert scale:\\
          Supported · Refuted · NEI · Conflicting Evidence
    \item Verdict = label with highest Likert score\\
          Ties: Refuted $>$ Supported $>$ CE $>$ NEI
    \item Returns reasoning trace, one-sentence justification, and key URLs
  \end{itemize}
\end{frame}

% ── 9. FIRE B ────────────────────────────────────────────────
\begin{frame}{Step 3 — FIRE Iterative Loop}

  After each verification round:

  \bigskip
  \begin{itemize}\setlength{\itemsep}{14pt}
    \item Compute \textbf{label margin} = top Likert score $-$ second-highest
    \item \textbf{Exit immediately} if verdict $\neq$ NEI and margin $\geq 4$
    \item \textbf{Always continue} if verdict = NEI — model declared insufficient evidence
    \item \textbf{Stop} if follow-up query is redundant (cosine sim $> 0.9$ vs.\ all past queries)
    \item Otherwise: generate targeted follow-up query, retrieve, merge, and repeat
  \end{itemize}

  \bigskip
  {\small\textcolor{slate}{Maximum 5 rounds.}}
\end{frame}

% ── 10. Likert rubric A ──────────────────────────────────────
\begin{frame}{Likert Rubric — Supported \& Refuted}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{forestg}{Supported}} — ``evidence confirms the claim''
      \begin{itemize}\setlength{\itemsep}{8pt}
        \item[\textbf{5}] Two+ credible sources ($\geq 0.6$) confirm all key qualifiers; no credible contradiction
        \item[\textbf{4}] Multiple sources agree, or one high-cred ($\geq 0.7$) source confirms directly
        \item[\textbf{3}] One credible ($\geq 0.5$) source supports; no corroboration
        \item[\textbf{1--2}] Weak or off-topic support only
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{crimson}{Refuted}} — ``evidence contradicts the claim''
      \begin{itemize}\setlength{\itemsep}{8pt}
        \item[\textbf{5}] Two+ credible sources ($\geq 0.6$) contradict a load-bearing qualifier; no credible support
        \item[\textbf{4}] One high-cred ($\geq 0.7$) source directly refutes a key element
        \item[\textbf{3}] One credible ($\geq 0.5$) source contradicts; no corroboration
        \item[\textbf{1--2}] Weak or off-topic contradiction only
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ── 11. Likert rubric B ──────────────────────────────────────
\begin{frame}{Likert Rubric — NEI \& Conflicting Evidence}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{slate}{Not Enough Evidence}}\\
      ``evidence does not address the claim''

      \bigskip
      \begin{itemize}\setlength{\itemsep}{8pt}
        \item[\textbf{5}] No retrieved item touches the claim at all
        \item[\textbf{3--4}] Topically related sources only; no direct address
        \item[\textbf{1--2}] At least one credible source ($\geq 0.5$) speaks to the substance — \emph{must not} score NEI above 2 in this case
      \end{itemize}

      \medskip
      {\small\textcolor{slate}{NEI is about coverage, not certainty.}}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{amber}{Conflicting Evidence}}\\
      ``credible sources meaningfully disagree''

      \bigskip
      \begin{itemize}\setlength{\itemsep}{8pt}
        \item[\textbf{5}] Two+ high-cred ($\geq 0.6$) sources on each side
        \item[\textbf{4}] One credible source supports AND one credible source refutes the same load-bearing element
        \item[\textbf{1--3}] All credible sources agree, or only minor wording differences
      \end{itemize}

      \medskip
      {\small\textcolor{slate}{If Supported and Refuted both $\geq 3$, raise CE to at least max(Supp,~Ref).}}
    \end{column}
  \end{columns}
\end{frame}

% ── 14. Misleadingness A ─────────────────────────────────────
\begin{frame}{Misleadingness Detection — The Problem}

  All SOTA systems target \emph{stated factual claims}: is what the post says true?
  None detect posts that are \alert{technically true but misleading.}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{14pt}
    \item \textbf{Misleading by scope}\\
          A broad claim backed only by a narrow instance.\\
          {\small ``A study shows vaccines cause autism'' — one retracted paper, not evidence.}
    \item \textbf{Misleading by implication}\\
          A true statement that naturally suggests a false conclusion.\\
          {\small ``Cities with more immigrants have higher crime rates'' — implies causation; omits confounders.}
    \item \textbf{Misleading by omission}\\
          True only under unstated conditions.
  \end{itemize}
\end{frame}

% ── 15. Misleadingness B ─────────────────────────────────────
\begin{frame}{Misleadingness Detection — The Algorithm}

  Applied after explicit claim verification. For each verified claim A:

  \bigskip
  \begin{enumerate}\setlength{\itemsep}{12pt}
    \item Establish A is true, preserving all qualifiers (time period, sample size, scope)
    \item \textbf{Strip qualifiers} $\to \tilde{A}$.
          If $\tilde{A}$ is false $\to$ \alert{misleading by scope}
    \item \textbf{Derive implication} B: the first factual inference a reasonable person draws from A.
          If B is false $\to$ \alert{misleading by implication}
    \item \textbf{Augment with silent assumptions} $\to A^+$.
          If $A^+$ is false $\to$ \alert{misleading by omission}
  \end{enumerate}

  \bigskip
  {\small Each derived claim is fed through the standard retrieval + verification sub-pipeline.
  A misleading verdict raises the post's falsity score even when the literal claim is \emph{Supported.}}

  \smallskip
  {\small\textcolor{slate}{Status: designed, not yet implemented.}}
\end{frame}

% ── 17. Error analysis ───────────────────────────────────────
\begin{frame}{Evaluation — Error Analysis}

  Of the 5 misses:

  \bigskip
  \begin{itemize}\setlength{\itemsep}{14pt}
    \item \textbf{Retrieval coverage gaps} (claims 3, 10)\\
          Regional or niche claims with no English web coverage.
          NEI is correct given what retrieval returned — the failure is upstream.
    \item \textbf{Date sensitivity} (claim 5)\\
          Verifier retrieved 2024 statistics for a 2020 claim and used parametric
          knowledge instead of period-appropriate evidence.\\
          {\small\textcolor{steel}{Fix: HerO 2 per-document summarisation with temporal anchoring.}}
    \item \textbf{Annotation disagreement} (claim 6)\\
          Four professional fact-checkers rated it Refuted; AVeriTeC gold = NEI on stricter wording.
          Genuine label conflict, not a pipeline error.
  \end{itemize}
\end{frame}

% ── 19. CE recall ────────────────────────────────────────────
\begin{frame}{Conflicting Evidence Recall = 0\%}

  12 gold-CE claims in the 200-claim run.
  Predicted: 6 Refuted, 5 NEI, 1 Supported, \alert{0 CE.}

  \medskip
  Under self-consistency ($\times 3$ runs), averaging Likert scores across rounds
  systematically suppresses CE: if runs 1 and 3 lean Refuted and run 2 leans CE,
  the averaged CE score rarely becomes the argmax.

  \bigskip
  \textbf{Fix candidates:}
  \begin{itemize}\setlength{\itemsep}{10pt}
    \item \textbf{Majority-vote aggregation} — one-line change; test this first
    \item \textbf{Targeted CE retrieval} — after Supported/Refuted, generate a query
          explicitly searching for the opposing view
    \item \textbf{Separate CE classifier} — fine-tuned on evidence pairs;
          bypasses LLM CE-avoidance bias
  \end{itemize}
\end{frame}

% ── 22. KB routing ───────────────────────────────────────────
\begin{frame}{Cost Lever: KB Routing Before Web Search}

  Currently all three sources (FCTAPI, Wikipedia, Serper) run in parallel on every claim.
  At scale, this means paying for Serper queries on claims professional fact-checkers have already verified.

  \bigskip
  \textbf{Fix: two-phase sequential routing}
  \begin{enumerate}\setlength{\itemsep}{8pt}
    \item Query FCTAPI + vector DB first {\small (free, $\sim$300\,ms)}
    \item KB hit $\to$ return verdict, skip Serper, FIRE exits after round 1
    \item No match $\to$ proceed to full Serper retrieval + FIRE
  \end{enumerate}

  \bigskip
  \textbf{Expected impact:} $\sim$40\% of novel claims resolved by KB in a mature system.
  KB hits carry credibility 0.90 — FIRE almost always exits after round 1.

  \bigskip
  {\small\textcolor{slate}{Code change: parallel \texttt{ThreadPoolExecutor} in \texttt{retrieval.py} becomes two-phase.}}
\end{frame}
